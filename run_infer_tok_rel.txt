upload_with_script() {
  local upload_script="$1"
  local label="$2"

  if [[ -f "${upload_script}" ]]; then
    echo "[upload] Uploading ${label} to Hugging Face..."
    python "${upload_script}"
    echo "[upload-done] ${label} uploaded to Hugging Face."
  else
    echo "Upload script does not exist: ${upload_script}" >&2
    exit 1
  fi
}
