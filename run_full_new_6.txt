fi
conda activate "${ENV_NAME}"

python -m pip install --upgrade pip
python -m pip install uv
uv sync
source .venv/bin/activate

HF_TOKEN="${HF_TOKEN:-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}"
hf auth login --force --token "${HF_TOKEN}"

  --filter part_6 \
  --gpus 6,7 \
  --gpus-per-job 2 \
  --infer-after-train \
  --infer-db full \
  --infer-batch-size 16
